    @type assistant_message() :: %{
  optional(:resume) => boolean(),
  optional(:content) => String.t(),
  optional(:thinking) => String.t(),
  optional(:tool_calls) => [tool_call()],
  role: String.t()
}

    @type command() ::
  :clear
  | :compact
  | {:resume, command()}
